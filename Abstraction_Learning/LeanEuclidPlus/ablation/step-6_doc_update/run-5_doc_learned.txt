-- Basic Geometric Sorts --
axiom Point : Type
axiom Line : Type
axiom Circle : Type



-- Inductive Types for Geometric Entities --
inductive Angle | right | ofPoints (A B C : Point)
inductive Segment | endpoints (a b : Point)
inductive Triangle | ofPoints (a b c : Point)



-- Notations and Macros for Geometric Entities --
"|(a─b)|" means the length of the line segment (type ℝ, not type Segment) between point a and point b.

"∠ a:b:c" means the degree of the angle (type ℝ, not type Angle) formed by points a, b, and c, where b is the vertex of the angle and a and c are points respectively on the two sides of the angle. Add parentheses around the angle notation like "(∠ a:b:c)" to avoid ambiguity.

"∟" means the degree of the right angle (type ℝ, not type Angle).

"△ a:b:c" means the triangle (type Triangle) formed from points a, b and c.
"Triangle.area △ a:b:c" means the area of the triangle formed by points a, b and c. Add parentheses around the triangle notation like "(△ a:b:c)" to avoid ambiguity.



-- Relations and Axioms for Geometric Sorts --
namespace Point
def onLine (a : Point) (L : Line) -- point a is on line L. The syntax is `a.onLine L`.

def sameSide (a b : Point) (L : Line) -- point a and b are on the same side of line L. They both are not on line L, but can be the same point. The syntax is `a.sameSide b L`.

def opposingSides (a b : Point) (L : Line) -- distinct point a and b are on opposite sides of line L. They both are not on line L. The syntax is `a.opposingSides b L`.

def onCircle (a: Point) (C: Circle) -- point a is on circle C. The syntax is `a.onCircle C`.

def insideCircle (a: Point) (C: Circle) -- point a is inside circle C. It can't be on the circle. The syntax is `a.insideCircle C`.

def outsideCircle (a: Point) (C: Circle)-- point a is outside circle C. It can't be on the circle. The syntax is `a.outsideCircle C`.

def isCentre (a: Point) (C: Circle) -- point a is on the unique center circle C. The syntax is `a.isCentre C`.
end Point


namespace Line
def intersectsLine (L M : Line) -- two lines L and M intersect at some point. They can be the same line. The syntax is `L.intersectsLine M`.

def intersectsCircle (L : Line) (C : Circle) -- line L and circle C intersect. The syntax is `L.intersectsCircle C`.
end Line


namespace Circle
def intersectsCircle (C1 C2: Circle) -- circle C1 and C2 intersect. The syntax is `C1.intersectsCircle C2`.
end Circle

end Triangle



-- Relations and Axioms for Geometric Sorts --
namespace Point
def isMidpointOf (m a b : Point) -- point m is the midpoint of segment AB iff b is between a and m and m and a are on the same line as b, and the distances from m to the endpoints are equal: between a m b ∧ |(a─m)| = |(m─b)|. The three points a, m, b are thus distinct and collinear. The syntax is `m.isMidpointOf a b`.

def isMidpointOfSegmentEndpoints (m a b : Point) (SP : Segment) -- same as `m.isMidpointOf a b`, with an explicit segment witness: SP must be exactly the segment with endpoints a and b, i.e. `SP = Segment.endpoints a b`. The syntax is `m.isMidpointOfSegmentEndpoints a b SP`.

def onExtensionBeyondB (p a b : Point) -- point p lies on the extension of segment AB beyond endpoint B iff B is between A and P, i.e. `between a b p`. The syntax is `p.onExtensionBeyondB a b`.

def onExtensionBeyondA (p a b : Point) -- point p lies on the extension of segment AB beyond endpoint A iff A is between B and P, i.e. `between b a p`. The syntax is `p.onExtensionBeyondA a b`.
end Point


namespace Triangle
def sideAB (a b : Point) (c : Point) -- the side AB of triangle ABC as a segment: `Segment.endpoints a b`. Note that the order of the first two point arguments determines the segment direction; the third point `c` is the remaining vertex of the triangle. The syntax is `Triangle.sideAB a b c`.

def sideBC (a : Point) (b c : Point) -- the side BC of triangle ABC as a segment: `Segment.endpoints b c`. The syntax is `Triangle.sideBC a b c`.

def sideCA (a : Point) (b : Point) (c : Point) -- the side CA of triangle ABC as a segment: `Segment.endpoints c a`. The syntax is `Triangle.sideCA a b c`.
end Triangle



-- Geometric Relations --
def distinctPointsOnLine (a b : Point) (L : Line) -- points a and b are distinct and on line L. The syntax is `distinctPointsOnLine a b L`.

def between (a b c : Point) -- mutually distinct points a, b and c are collinear and ordered. Point b is between point a and c. The syntax is `between a b c`.

def formTriangle (a b c : Point) (AB BC CA : Line) -- mutually distinct points a, b and c form a triangle, where point a and b are on line AB, point b and c are on line BC, point a and c are on line CA. The lines AB, BC, and CA must be distinct. Note that the order and correspondence of arguments a, b, c, AB, BC, CA is strictly required i.e. a and b must be on AB, b and c must be on BC, a and c must be on CA, and they must be passed in the exact order! The syntax is `formTriangle a b c AB BC CA`.

def formRectilinearAngle (a b c : Point) (AB BC : Line)  -- points a, b and c form a rectilinear angle, where b is the vertex of the angle and a and c are points respectively on the sides AB and BC. The sides AB and BC can be the same line, the points a and c can be the same point, but the vertex b must be distinct from a and c. The syntax is `formRectilinearAngle a b c AB BC`.

def formParallelogram (a b c d : Point) (AB CD AC BD : Line) -- mutually distinct points a, b, d, and c (in clockwise/counterclockwise order i.e. ad is a diagonal) form a parallelogram, where points a and b are on line AB, points c and d are on line CD, points a and c are on line AC, and points b and d are on line BD. The lines AB, CD, AC, and BD must be distinct. Note that the order and correspondence of arguments a, b, c, d, AB, CD, AC, BD is strictly required i.e. a and b must be on AB, c and d must be on CD, a and c must be on AC, b and d must be on BD, and they must be passed in the exact order! The syntax is `formParallelogram a b c d AB CD AC BD`.

def twoDistinctLinesIntersectAtPoint (L M : Line) (i : Point) -- two distinct lines L and M intersect at point i iff L ≠ M, i lies on both L and M, and L.intersectsLine M holds. This records both the witness point and the non-coincidence of the lines. The syntax is `twoDistinctLinesIntersectAtPoint L M i`.

def sequentiallyAligned (pts : List Point) -- a list of points is sequentially aligned when consecutive triples follow the betweenness order (ordered collinear chain). There is no explicit line or global distinctness requirement beyond the betweenness constraints of each triple. Example: `sequentiallyAligned [a, b, c, d]` means between a b c and between b c d. The syntax is `sequentiallyAligned [p₁, p₂, ..., pₙ]` with n ≥ 0.

def sequentiallyAlignedThreeOrMore (pts : List Point) -- same as `sequentiallyAligned pts` but additionally requires at least three points in the list. The syntax is `sequentiallyAlignedThreeOrMore [p₁, p₂, p₃, ...]`.

def twoSetsOpposingSidesOnLine (xs ys : List Point) (L : Line) -- two lists of points lie on opposing sides of line L iff: (1) no point from either list is on L; and (2) every cross-pair (x ∈ xs, y ∈ ys) is on opposite sides of L. There is no requirement that xs and ys be disjoint or nonempty. The syntax is `twoSetsOpposingSidesOnLine [x₁, ..., x_m] [y₁, ..., y_n] L`.

def supplementaryAngles (a b c d e f : Point) -- the angles ∠ a:b:c and ∠ d:e:f are supplementary iff (∠ a:b:c) + (∠ d:e:f) = ∟ + ∟. The angle points are ordered as usual with the middle point the vertex. The syntax is `supplementaryAngles a b c d e f`.

def parallel (L M : Line) -- lines L and M are parallel iff L ≠ M and they do not intersect, i.e. ¬ L.intersectsLine M. Note this is a relation on lines (no namespace). The syntax is `parallel L M`.

def nonCollinearPoints (a b c : Point) -- points a, b, c are non-collinear iff the area of triangle ABC is nonzero, i.e. Triangle.area (△ a:b:c) ≠ 0. The syntax is `nonCollinearPoints a b c`.

def segmentFromVertexToOppositeSide (v s1 s2 p : Point) (SP : Segment) -- the segment SP connects vertex v to a point p on the opposite side s1s2 of a triangle, witnessed by the betweenness relation `between s1 p s2`. Formally: SP = Segment.endpoints v p ∧ between s1 p s2. No further triangle-formation checks are imposed here. The syntax is `segmentFromVertexToOppositeSide v s1 s2 p SP`.

def congruentSegments (a b c d : Point) -- segments AB and CD are congruent iff their lengths are equal: |AB| = |CD|. The syntax is `congruentSegments a b c d`.

def perpendicularAt (L M : Line) (i a c : Point) -- lines L and M are perpendicular at witness point i via points a on L and c on M iff: (1) L and M intersect; (2) a, i, c form a rectilinear angle with sides L and M; and (3) ∠ AIC is a right angle, i.e. (∠ a:i:c) = ∟. The witness a lies on L, c lies on M, and i is the vertex. The syntax is `perpendicularAt L M i a c`.

def equalLengthRatios (a b c d e f g h : Point) -- equality of two ratios of segment lengths: (|AB| / |CD|) = (|EF| / |GH|). No non-degeneracy conditions are enforced here (denominators may be zero in ℝ); add side conditions if needed. The syntax is `equalLengthRatios a b c d e f g h`.

def triangleAngleSum (a b c : Point) -- for a non-degenerate triangle ABC, the sum of its interior angles is ∟ + ∟ (i.e., 180°): nonCollinearPoints a b c ∧ (∠ b:a:c + ∠ a:b:c + ∠ a:c:b = ∟ + ∟). The syntax is `triangleAngleSum a b c`.

def trianglesSimilar (a b c d e f : Point) -- triangles ABC and DEF are similar iff the three corresponding angles are equal and the corresponding sides are in proportion: ∠ BAC = ∠ EDF, ∠ ABC = ∠ DEF, ∠ ACB = ∠ DFE and |AB|/|DE| = |BC|/|EF| = |CA|/|FD| (encoded by two equalities). No non-degeneracy assumptions are enforced. The vertex correspondence is A↔D, B↔E, C↔F in this exact order. The syntax is `trianglesSimilar a b c d e f`.

def trianglesCongruent (a b c d e f : Point) -- triangles ABC and DEF are congruent iff the three corresponding angles are equal and the three corresponding sides are equal: A↔D, B↔E, C↔F in this exact order. The syntax is `trianglesCongruent a b c d e f`.

def formConvexQuadrilateral (a b c d : Point) (AB BC CD DA : Line) -- four points a, b, c, d with side-lines AB, BC, CD, DA form a convex quadrilateral iff: (1) a and b are distinct and on AB; (2) b and c are distinct and on BC; (3) c and d are distinct and on CD; (4) d and a are distinct and on DA; and (5) each side line separates the two nonincident vertices on the same side, namely: c and d are on the same side of AB; d and a are on the same side of BC; a and b are on the same side of CD; b and c are on the same side of DA. No extra distinctness, non-collinearity, or intersection requirements are added beyond these primitives. The order and correspondence is strict: (a,b)↔AB, (b,c)↔BC, (c,d)↔CD, (d,a)↔DA must be passed in exactly this order. The syntax is `formConvexQuadrilateral a b c d AB BC CD DA`.

def diagonalACOfQuadrilateral (a b c d : Point) (SP : Segment) -- SP is the diagonal AC of the ordered quadruple (a, b, c, d), i.e. SP = Segment.endpoints a c. The non-diagonal points b and d are ignored but their positions matter in the ordered quadruple. The syntax is `diagonalACOfQuadrilateral a b c d SP`.

def diagonalBDOfQuadrilateral (a b c d : Point) (SP : Segment) -- SP is the diagonal BD of the ordered quadruple (a, b, c, d), i.e. SP = Segment.endpoints b d. The syntax is `diagonalBDOfQuadrilateral a b c d SP`.

def segmentBisectsAngleAtVertex (a b c p : Point) (SP : Segment) (AB BC BP : Line) -- segment BP bisects angle ABC at vertex b with witnesses: a,b on AB; b,c on BC; b,p on BP; interior witnessed by same-side constraints (a and p are on the same side of BC; c and p are on the same side of AB); the sub-angles satisfy ∠ ABP = ∠ PBC; and SP is exactly the segment with endpoints b and p. The order is strict: angle is ABC (vertex b); AB, BC are its sides; BP is the bisector carrier line; SP = [b,p]. The syntax is `segmentBisectsAngleAtVertex a b c p SP AB BC BP`.

def trianglesShareSideBy
  (sel₁ : Point → Point → Point → Segment)
  (sel₂ : Point → Point → Point → Segment)
  (a b c d e f : Point) -- triangles ABC and DEF share a side when the two selected side segments are equal as segments. Typical selectors are `Triangle.sideAB`, `Triangle.sideBC`, `Triangle.sideCA`. The vertex correspondence for each selector must be respected. The syntax is `trianglesShareSideBy sel₁ sel₂ a b c d e f`.

def equilateralTriangle (a b c : Point) -- triangle ABC is equilateral iff |AB| = |BC| = |CA| (encoded as |AB| = |BC| and |BC| = |CA|). No non-degeneracy is enforced. The syntax is `equilateralTriangle a b c`.

def isoscelesAtA (a b c : Point) -- triangle ABC is isosceles at vertex A iff |AB| = |AC|. The syntax is `isoscelesAtA a b c`.

def isoscelesAtB (a b c : Point) -- triangle ABC is isosceles at vertex B iff |BA| = |BC|. The syntax is `isoscelesAtB a b c`.

def isoscelesAtC (a b c : Point) -- triangle ABC is isosceles at vertex C iff |CA| = |CB|. The syntax is `isoscelesAtC a b c`.

def lineCutsTriangleOnABandACAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AB and AC of triangle ABC at points p and q respectively iff: formTriangle a b c AB BC CA; between a p b; between c q a; p.onLine L; q.onLine L. The order and correspondence are strict: (a,b) on AB, (b,c) on BC, (c,a) on CA; p is on AB between A and B; q is on AC between C and A. The syntax is `lineCutsTriangleOnABandACAt a b c p q AB BC CA L`.

def lineCutsTriangleOnABandBCAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AB and BC at p and q respectively: formTriangle a b c AB BC CA; between a p b; between b q c; p.onLine L; q.onLine L. The order and correspondence are strict. The syntax is `lineCutsTriangleOnABandBCAt a b c p q AB BC CA L`.

def lineCutsTriangleOnACandBCAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AC and BC at p and q respectively: formTriangle a b c AB BC CA; between c p a; between b q c; p.onLine L; q.onLine L. The order and correspondence are strict. The syntax is `lineCutsTriangleOnACandBCAt a b c p q AB BC CA L`.

def lineCutsABandACParallelBC (a b c p q : Point) (AB BC CA L : Line) -- line L cuts sides AB and AC at p and q as above and is parallel to BC: same requirements as `lineCutsTriangleOnABandACAt ...` together with `parallel L BC`. The syntax is `lineCutsABandACParallelBC a b c p q AB BC CA L`.

def lineCutsABandBCParallelCA (a b c p q : Point) (AB BC CA L : Line) -- line L cuts sides AB and BC at p and q and is parallel to CA: same requirements as `lineCutsTriangleOnABandBCAt ...` together with `parallel L CA`. The syntax is `lineCutsABandBCParallelCA a b c p q AB BC CA L`.

def lineCutsACandBCParallelAB (a b c p q : Point) (AB BC CA L : Line) -- line L cuts sides AC and BC at p and q and is parallel to AB: same requirements as `lineCutsTriangleOnACandBCAt ...` together with `parallel L AB`. The syntax is `lineCutsACandBCParallelAB a b c p q AB BC CA L`.
