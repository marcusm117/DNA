-- Basic Geometric Sorts --
axiom Point : Type
axiom Line : Type
axiom Circle : Type



-- Inductive Types for Geometric Entities --
inductive Angle | right | ofPoints (A B C : Point)
inductive Segment | endpoints (a b : Point)
inductive Triangle | ofPoints (a b c : Point)



-- Notations and Macros for Geometric Entities --
"|(a─b)|" means the length of the line segment (type ℝ, not type Segment) between point a and point b.

"∠ a:b:c" means the degree of the angle (type ℝ, not type Angle) formed by points a, b, and c, where b is the vertex of the angle and a and c are points respectively on the two sides of the angle. Add parentheses around the angle notation like "(∠ a:b:c)" to avoid ambiguity.

"∟" means the degree of the right angle (type ℝ, not type Angle).

"△ a:b:c" means the triangle (type Triangle) formed from points a, b and c.
"Triangle.area △ a:b:c" means the area of the triangle formed by points a, b and c. Add parentheses around the triangle notation like "(△ a:b:c)" to avoid ambiguity.



-- Relations and Axioms for Geometric Sorts --
namespace Point
def onLine (a : Point) (L : Line) -- point a is on line L. The syntax is `a.onLine L`.

def sameSide (a b : Point) (L : Line) -- point a and b are on the same side of line L. They both are not on line L, but can be the same point. The syntax is `a.sameSide b L`.

def opposingSides (a b : Point) (L : Line) -- distinct point a and b are on opposite sides of line L. They both are not on line L. The syntax is `a.opposingSides b L`.

def onCircle (a: Point) (C: Circle) -- point a is on circle C. The syntax is `a.onCircle C`.

def insideCircle (a: Point) (C: Circle) -- point a is inside circle C. It can't be on the circle. The syntax is `a.insideCircle C`.

def outsideCircle (a: Point) (C: Circle)-- point a is outside circle C. It can't be on the circle. The syntax is `a.outsideCircle C`.

def isCentre (a: Point) (C: Circle) -- point a is on the unique center circle C. The syntax is `a.isCentre C`.
end Point


namespace Line
def intersectsLine (L M : Line) -- two lines L and M intersect at some point. They can be the same line. The syntax is `L.intersectsLine M`.

def intersectsCircle (L : Line) (C : Circle) -- line L and circle C intersect. The syntax is `L.intersectsCircle C`.
end Line


namespace Circle
def intersectsCircle (C1 C2: Circle) -- circle C1 and C2 intersect. The syntax is `C1.intersectsCircle C2`.
end Circle


namespace Triangle
end Triangle


-- Geometric Relations --
def distinctPointsOnLine (a b : Point) (L : Line) -- points a and b are distinct and on line L. The syntax is `distinctPointsOnLine a b L`.

def between (a b c : Point) -- mutually distinct points a, b and c are collinear and ordered. Point b is between point a and c. The syntax is `between a b c`.

def formTriangle (a b c : Point) (AB BC CA : Line) -- mutually distinct points a, b and c form a triangle, where point a and b are on line AB, point b and c are on line BC, point a and c are on line CA. The lines AB, BC, and CA must be distinct. Note that the order and correspondence of arguments a, b, c, AB, BC, CA is strictly required i.e. a and b must be on AB, b and c must be on BC, a and c must be on CA, and they must be passed in the exact order! The syntax is `formTriangle a b c AB BC CA`.

def formRectilinearAngle (a b c : Point) (AB BC : Line)  -- points a, b and c form a rectilinear angle, where b is the vertex of the angle and a and c are points respectively on the sides AB and BC. The sides AB and BC can be the same line, the points a and c can be the same point, but the vertex b must be distinct from a and c. The syntax is `formRectilinearAngle a b c AB BC`.

def formParallelogram (a b c d : Point) (AB CD AC BD : Line) -- mutually distinct points a, b, d, and c (in clockwise/counterclockwise order i.e. ad is a diagonal) form a parallelogram, where points a and b are on line AB, points c and d are on line CD, points a and c are on line AC, and points b and d are on line BD. The lines AB, CD, AC, and BD must be distinct. Note that the order and correspondence of arguments a, b, c, d, AB, CD, AC, BD is strictly required i.e. a and b must be on AB, c and d must be on CD, a and c must be on AC, b and d must be on BD, and they must be passed in the exact order! The syntax is `formParallelogram a b c d AB CD AC BD`.



-- Relations and Axioms for Geometric Sorts --
namespace Point
def isMidpointOf (m a b : Point) -- point m is the midpoint of segment AB iff m lies between a and b and |AM| = |MB|. The syntax is `m.isMidpointOf a b`.

def isMidpointOfSegmentEndpoints (m a b : Point) (SP : Segment) -- same as `m.isMidpointOf a b` but additionally witnesses the carrier segment: `SP = Segment.endpoints a b`. The syntax is `m.isMidpointOfSegmentEndpoints a b SP`.

def onExtensionBeyondB (p a b : Point) -- point p lies on the extension of segment AB beyond endpoint B iff `between a b p`. The syntax is `p.onExtensionBeyondB a b`.

def onExtensionBeyondA (p a b : Point) -- point p lies on the extension of segment AB beyond endpoint A iff `between b a p`. The syntax is `p.onExtensionBeyondA a b`.
end Point


namespace Triangle
def sideAB (a b c : Point) -- the side AB of triangle ABC as a segment i.e. `Segment.endpoints a b`. The syntax is `Triangle.sideAB a b c`.

def sideBC (a b c : Point) -- the side BC of triangle ABC as a segment i.e. `Segment.endpoints b c`. The syntax is `Triangle.sideBC a b c`.

def sideCA (a b c : Point) -- the side CA of triangle ABC as a segment i.e. `Segment.endpoints c a`. The syntax is `Triangle.sideCA a b c`.
end Triangle



-- Geometric Relations --
def twoDistinctLinesIntersectAtPoint (L M : Line) (i : Point) -- two distinct lines L and M intersect at point i, i lies on both L and M, and L.intersectsLine M holds. The syntax is `twoDistinctLinesIntersectAtPoint L M i`.

def sequentiallyAligned (pts : List Point) -- a list of points is aligned in order via chained betweenness across consecutive triples; no extra global line or distinctness constraints are added beyond those enforced by betweenness. The syntax is `sequentiallyAligned [a, b, c, d]`.

def sequentiallyAlignedThreeOrMore (pts : List Point) -- same as `sequentiallyAligned pts` and additionally requires the list to contain at least three points. The syntax is `sequentiallyAlignedThreeOrMore [a, b, c, d]`.

def twoSetsOpposingSidesOnLine (xs ys : List Point) (L : Line) -- every point in xs and ys is not on line L, and every cross-pair (x in xs, y in ys) is on opposite sides of L. The syntax is `twoSetsOpposingSidesOnLine [a₁, a₂] [b₁, b₂, b₃] L`.

def supplementaryAngles (a b c d e f : Point) -- angles ∠ a:b:c and ∠ d:e:f are supplementary iff their measures sum to ∟ + ∟. The syntax is `supplementaryAngles a b c d e f`.

def parallel (L M : Line) -- lines L and M are parallel iff L ≠ M and ¬ L.intersectsLine M. The syntax is `parallel L M`.

def nonCollinearPoints (a b c : Point) -- points a, b, c are non-collinear iff Triangle.area (△ a:b:c) ≠ 0. The syntax is `nonCollinearPoints a b c`.

def segmentFromVertexToOppositeSide (v s1 s2 p : Point) (SP : Segment) -- a segment SP connects vertex v to a point p on the opposite side s1s2 of a triangle, witnessed by betweenness `between s1 p s2`. Requires SP = Segment.endpoints v p. The syntax is `segmentFromVertexToOppositeSide v s1 s2 p SP`.

def congruentSegments (a b c d : Point) -- segments AB and CD are congruent iff |AB| = |CD|. The syntax is `congruentSegments a b c d`.

def perpendicularAt (L M : Line) (i a c : Point) -- lines L and M are perpendicular with witness at point i using points a on L and c on M iff: L and M intersect, a,i,c form a rectilinear angle with sides on L and M, and ∠ a:i:c = ∟. The syntax is `perpendicularAt L M i a c`.

def equalLengthRatios (a b c d e f g h : Point) -- the ratios of segment lengths are equal: (|AB| / |CD|) = (|EF| / |GH|). No separate non-zero denominator guard is imposed here. The syntax is `equalLengthRatios a b c d e f g h`.

def triangleAngleSum (a b c : Point) -- for a non-degenerate triangle ABC (i.e. nonCollinearPoints a b c), the interior angle sum holds: ∠ BAC + ∠ ABC + ∠ ACB = ∟ + ∟. The syntax is `triangleAngleSum a b c`.

def trianglesSimilar (a b c d e f : Point) -- triangles ABC and DEF are similar iff: (i) corresponding angles are equal: ∠ BAC = ∠ EDF, ∠ ABC = ∠ DEF, ∠ ACB = ∠ DFE; and (ii) corresponding side-length ratios are equal: |AB|/|DE| = |BC|/|EF| and |BC|/|EF| = |CA|/|FD|. The correspondence and order of arguments is strict: (a,b,c) correspond to (d,e,f) respectively. The syntax is `trianglesSimilar a b c d e f`.

def trianglesCongruent (a b c d e f : Point) -- triangles ABC and DEF are congruent iff their three corresponding angles are equal and their three corresponding sides are pairwise equal in length. The correspondence and order of arguments is strict: (a,b,c) correspond to (d,e,f) respectively. The syntax is `trianglesCongruent a b c d e f`.

def formConvexQuadrilateral (a b c d : Point) (AB BC CD DA : Line) -- points a, b, c, d with side-lines AB, BC, CD, DA form a convex quadrilateral iff: (i) consecutive vertices lie on their side lines with distinct endpoints (`distinctPointsOnLine a b AB`, `distinctPointsOnLine b c BC`, `distinctPointsOnLine c d CD`, `distinctPointsOnLine d a DA`); and (ii) for each side line, the two nonincident vertices lie on the same side of that line (`c.sameSide d AB`, `d.sameSide a BC`, `a.sameSide b CD`, `b.sameSide c DA`). No additional conditions (like pairwise distinct side lines, extra intersection constraints, or explicit non-collinearity) are imposed beyond those already enforced by these primitives. The ordered correspondence is strict: a and b must be on AB, b and c on BC, c and d on CD, and d and a on DA, in exactly this order. The syntax is `formConvexQuadrilateral a b c d AB BC CD DA`.

def diagonalACOfQuadrilateral (a b c d : Point) (SP : Segment) -- SP is the AC diagonal of the ordered quadruple (a, b, c, d), i.e. `SP = Segment.endpoints a c`. The syntax is `diagonalACOfQuadrilateral a b c d SP`.

def diagonalBDOfQuadrilateral (a b c d : Point) (SP : Segment) -- SP is the BD diagonal of the ordered quadruple (a, b, c, d), i.e. `SP = Segment.endpoints b d`. The syntax is `diagonalBDOfQuadrilateral a b c d SP`.

def segmentBisectsAngleAtVertex (a b c p : Point) (SP : Segment) (AB BC BP : Line) -- segment BP bisects angle ABC at vertex b, with side lines AB and BC and carrier line BP for the bisector, requiring: a,b on AB; b,c on BC; b,p on BP; a and p on the same side of BC; c and p on the same side of AB; the two sub-angles are equal, ∠ ABP = ∠ PBC; and SP = Segment.endpoints b p. The order and correspondence of (a,b,c,p,AB,BC,BP,SP) is strict. The syntax is `segmentBisectsAngleAtVertex a b c p SP AB BC BP`.

def trianglesShareSideBy (sel₁ sel₂ : Point → Point → Point → Segment) (a b c d e f : Point) -- triangles ABC and DEF share a side when the selected side segments are equal, i.e. `sel₁ a b c = sel₂ d e f`. Typical selectors are `Triangle.sideAB`, `Triangle.sideBC`, `Triangle.sideCA`. The syntax is `trianglesShareSideBy Triangle.sideAB Triangle.sideBC a b c d e f`.

def equilateralTriangle (a b c : Point) -- triangle ABC is equilateral iff |AB| = |BC| and |BC| = |CA|. The syntax is `equilateralTriangle a b c`.

def isoscelesAtA (a b c : Point) -- triangle ABC is isosceles at vertex A iff |AB| = |AC|. The syntax is `isoscelesAtA a b c`.

def isoscelesAtB (a b c : Point) -- triangle ABC is isosceles at vertex B iff |BA| = |BC|. The syntax is `isoscelesAtB a b c`.

def isoscelesAtC (a b c : Point) -- triangle ABC is isosceles at vertex C iff |CA| = |CB|. The syntax is `isoscelesAtC a b c`.

def lineCutsTriangleOnABandACAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AB and AC of triangle ABC at points p and q respectively, with: `formTriangle a b c AB BC CA`, `between a p b` (p on AB), `between c q a` (q on AC), and both p,q on L. The syntax is `lineCutsTriangleOnABandACAt a b c p q AB BC CA L`.

def lineCutsTriangleOnABandBCAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AB and BC at points p and q respectively, with: `formTriangle a b c AB BC CA`, `between a p b` (p on AB), `between b q c` (q on BC), and both p,q on L. The syntax is `lineCutsTriangleOnABandBCAt a b c p q AB BC CA L`.

def lineCutsTriangleOnACandBCAt (a b c p q : Point) (AB BC CA L : Line) -- line L intersects sides AC and BC at points p and q respectively, with: `formTriangle a b c AB BC CA`, `between c p a` (p on AC), `between b q c` (q on BC), and both p,q on L. The syntax is `lineCutsTriangleOnACandBCAt a b c p q AB BC CA L`.

def lineCutsABandACParallelBC (a b c p q : Point) (AB BC CA L : Line) -- line L cuts AB at p and AC at q and is parallel to BC, with all requirements from `lineCutsTriangleOnABandACAt` plus `parallel L BC`. The syntax is `lineCutsABandACParallelBC a b c p q AB BC CA L`.

def lineCutsABandBCParallelCA (a b c p q : Point) (AB BC CA L : Line) -- line L cuts AB at p and BC at q and is parallel to CA, with all requirements from `lineCutsTriangleOnABandBCAt` plus `parallel L CA`. The syntax is `lineCutsABandBCParallelCA a b c p q AB BC CA L`.

def lineCutsACandBCParallelAB (a b c p q : Point) (AB BC CA L : Line) -- line L cuts AC at p and BC at q and is parallel to AB, with all requirements from `lineCutsTriangleOnACandBCAt` plus `parallel L AB`. The syntax is `lineCutsACandBCParallelAB a b c p q AB BC CA L`.
